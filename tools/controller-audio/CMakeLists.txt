cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(controller_audio_receiver LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
if(NOT TARGET SDL3::SDL3)
 find_package(SDL3 3.2 CONFIG REQUIRED)
endif()
get_filename_component(KYTY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(controller_audio_receiver_core STATIC receiver.cpp usbSink.cpp "${KYTY_ROOT}/src/controller_audio/protocol.cpp")
target_include_directories(controller_audio_receiver_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${KYTY_ROOT}/src" "${KYTY_ROOT}/3rdparty/nlohmann_json/include")
target_link_libraries(controller_audio_receiver_core PUBLIC OpenSSL::Crypto)
add_executable(controller-audio-receiver main.cpp transport.cpp usbOutput.cpp)
target_link_libraries(controller-audio-receiver PRIVATE controller_audio_receiver_core OpenSSL::SSL SDL3::SDL3)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 include(CTest)
endif()
if(BUILD_TESTING)
 add_executable(controller_audio_receiver_tests "${KYTY_ROOT}/tests/ControllerAudioReceiverTests.cpp")
 target_link_libraries(controller_audio_receiver_tests PRIVATE controller_audio_receiver_core)
 add_test(NAME controller_audio_receiver_pure COMMAND controller_audio_receiver_tests)
 add_library(controller_audio_no_hardware SHARED "${KYTY_ROOT}/tests/ControllerAudioReceiverNoHardware.cpp")
 target_link_libraries(controller_audio_no_hardware PRIVATE ${CMAKE_DL_LIBS})
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME controller_audio_receiver_tls_null COMMAND "${Python3_EXECUTABLE}" "${KYTY_ROOT}/tests/ControllerAudioReceiverLoopback.py" "$<TARGET_FILE:controller-audio-receiver>" "$<TARGET_FILE:controller_audio_no_hardware>")
 set_tests_properties(controller_audio_receiver_tls_null PROPERTIES TIMEOUT 20)
 add_test(NAME controller_audio_receiver_installer COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_install.py")
 set_tests_properties(controller_audio_receiver_installer PROPERTIES TIMEOUT 30)
endif()
install(TARGETS controller-audio-receiver RUNTIME DESTINATION bin)
