cmake_minimum_required(VERSION 3.16)
project(kyty_companion_bluetooth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(KYTY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(KYTY_SDL_INCLUDE "" CACHE PATH "Directory containing SDL3/SDL.h")
set(KYTY_OPUS_INCLUDE "" CACHE PATH "Directory containing opus.h")
if(NOT KYTY_SDL_INCLUDE)
  find_path(KYTY_SDL_INCLUDE_FOUND SDL3/SDL.h REQUIRED)
  set(KYTY_SDL_INCLUDE "${KYTY_SDL_INCLUDE_FOUND}")
endif()
if(NOT KYTY_OPUS_INCLUDE)
  find_path(KYTY_OPUS_INCLUDE_FOUND opus.h PATH_SUFFIXES opus REQUIRED)
  set(KYTY_OPUS_INCLUDE "${KYTY_OPUS_INCLUDE_FOUND}")
endif()
find_library(KYTY_SDL_LIBRARY NAMES SDL3 libSDL3.so.0 REQUIRED)
find_library(KYTY_OPUS_LIBRARY NAMES opus libopus.so.0 REQUIRED)
find_package(Threads REQUIRED)
add_library(kyty_original_bt OBJECT "${KYTY_ROOT}/src/libs/dualSenseBluetooth.cpp")
set_target_properties(kyty_original_bt PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(kyty_original_bt PRIVATE shim "${KYTY_ROOT}/src" "${KYTY_SDL_INCLUDE}" "${KYTY_OPUS_INCLUDE}")
target_compile_definitions(kyty_original_bt PRIVATE SDL_hid_write=kyty_bt_hid_write SDL_hid_open_path=kyty_bt_hid_open_path SDL_hid_close=kyty_bt_hid_close)
add_library(kyty_companion_bt SHARED bridge.cpp $<TARGET_OBJECTS:kyty_original_bt>)
target_include_directories(kyty_companion_bt PRIVATE "${KYTY_ROOT}/src" "${KYTY_SDL_INCLUDE}")
target_link_libraries(kyty_companion_bt PRIVATE "${KYTY_SDL_LIBRARY}" "${KYTY_OPUS_LIBRARY}" Threads::Threads)
set_target_properties(kyty_companion_bt PROPERTIES OUTPUT_NAME kyty_companion_bt)
file(SHA256 "${KYTY_ROOT}/src/libs/dualSenseBluetooth.cpp" KYTY_BT_SOURCE_SHA256)
target_compile_definitions(kyty_companion_bt PRIVATE KYTY_BT_SOURCE_SHA256="${KYTY_BT_SOURCE_SHA256}")
include(CTest)
if(BUILD_TESTING)
  add_executable(test_native test_native.cpp bridge.cpp $<TARGET_OBJECTS:kyty_original_bt>)
  target_include_directories(test_native PRIVATE "${KYTY_ROOT}/src" "${KYTY_SDL_INCLUDE}" "${KYTY_OPUS_INCLUDE}")
  target_compile_definitions(test_native PRIVATE KYTY_BT_SOURCE_SHA256="${KYTY_BT_SOURCE_SHA256}")
  target_link_libraries(test_native PRIVATE "${KYTY_SDL_LIBRARY}" "${KYTY_OPUS_LIBRARY}" Threads::Threads)
  foreach(symbol SDL_LockJoysticks SDL_UnlockJoysticks SDL_CreateAudioStream SDL_GetGamepads SDL_GetGamepadFromID SDL_GetGamepadType SDL_GetGamepadConnectionState SDL_GetGamepadVendor SDL_GetGamepadProduct SDL_GetGamepadSerial SDL_SendGamepadEffect SDL_RumbleGamepad SDL_hid_enumerate SDL_hid_free_enumeration SDL_hid_open_path SDL_hid_close SDL_hid_write)
    target_link_options(test_native PRIVATE "-Wl,--wrap=${symbol}")
  endforeach()
  add_test(NAME native_fake_controller COMMAND test_native)
endif()
