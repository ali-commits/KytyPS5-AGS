cmake_minimum_required(VERSION 3.22)
project(AgsRustStreamingControls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test Concurrent Network)
add_executable(ags_rust_controls_tests test.cpp ../../src/agsRustStreamingDialog.cpp ../../src/hostControlClient.cpp ../../src/hostSocket.cpp ../../include/hostControlClient.h ../../include/hostSocket.h)
target_include_directories(ags_rust_controls_tests PRIVATE ../../include)
target_link_libraries(ags_rust_controls_tests PRIVATE Qt6::Widgets Qt6::Test Qt6::Concurrent Qt6::Network)
enable_testing()
add_test(NAME rust_controls COMMAND ags_rust_controls_tests)
set_tests_properties(rust_controls PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
